-- ============================================================ -- FIX — get_trip_settlement: ambiguous "id" column (v2) -- Run once in Supabase → SQL Editor. -- ============================================================ -- Bug: -- The previous version did jsonb_to_recordset(...) + a lateral select -- that re-projected "id", "name", "net" alongside columns that already -- carried those names from the source. Postgres complained: -- column reference "id" is ambiguous -- -- Fix: -- Skip the JSON round-trip entirely. Compute paid/share/net in a CTE -- and read it straight into parallel arrays for the greedy transfer -- loop. Cleaner and faster. create or replace function public.get_trip_settlement(p_trip uuid) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_uid uuid := auth.uid(); v_total numeric := 0; v_balances jsonb; v_transfers jsonb := '[]'::jsonb; arr_id uuid[]; arr_name text[]; arr_net numeric[]; i int; j int; n int; v_amt numeric; v_eps numeric := 0.005; -- treat sub-cent residuals as zero begin if v_uid is null then raise exception 'Not signed in.'; end if; if not public.is_global_admin() and not public.is_trip_member(p_trip) then raise exception 'No access to this trip.'; end if; -- Build paid / share / net per trip member, in ONE pass, then read it -- into both the JSON output AND parallel arrays for the transfer loop. create temp table tmp_bals on commit drop as with members as ( select tm.directory_id as id, coalesce(nullif(trim(d.first_name||' '||d.last_name),''), d.email, 'Unnamed') as name from public.trip_members tm join public.directory d on d.id = tm.directory_id where tm.trip_id = p_trip ), paid as ( select e.paid_by as id, sum(e.amount) as amt from public.expenses e where e.trip_id = p_trip and e.status = 'active' group by e.paid_by ), share as ( select es.directory_id as id, sum(e.amount / nullif(( select count(*) from public.expense_splits s where s.expense_id = e.id ),0)) as amt from public.expense_splits es join public.expenses e on e.id = es.expense_id where e.trip_id = p_trip and e.status = 'active' group by es.directory_id ) select m.id as dir_id, m.name as dir_name, round(coalesce(p.amt,0)::numeric, 2) as paid, round(coalesce(s.amt,0)::numeric, 2) as share, round((coalesce(p.amt,0) - coalesce(s.amt,0))::numeric, 2) as net from members m left join paid p on p.id = m.id left join share s on s.id = m.id; -- Trip total select coalesce(sum(amount),0)::numeric into v_total from public.expenses where trip_id = p_trip and status = 'active'; -- Balances JSON (sorted: biggest creditors first looks nicer in UI) select coalesce(jsonb_agg(jsonb_build_object( 'directory_id', dir_id, 'name', dir_name, 'paid', paid, 'share', share, 'net', net ) order by net desc), '[]'::jsonb) into v_balances from tmp_bals; -- Parallel arrays for the transfer loop, sorted ascending by net so -- the biggest debtor (most negative) is at index 1. select array_agg(dir_id order by net asc), array_agg(dir_name order by net asc), array_agg(net order by net asc) into arr_id, arr_name, arr_net from tmp_bals; -- Greedy transfer matching: pair the biggest debtor with the biggest -- creditor, settle as much as possible, move on. if arr_net is not null then n := array_length(arr_net,1); i := 1; j := n; while i is not null and j is not null and i < j loop if arr_net[i] > -v_eps then exit; end if; -- no real debt left if arr_net[j] < v_eps then exit; end if; -- no real credit left v_amt := round(least(-arr_net[i], arr_net[j])::numeric, 2); if v_amt > 0 then v_transfers := v_transfers || jsonb_build_array(jsonb_build_object( 'from_id', arr_id[i], 'from_name', arr_name[i], 'to_id', arr_id[j], 'to_name', arr_name[j], 'amount', v_amt )); arr_net[i] := arr_net[i] + v_amt; arr_net[j] := arr_net[j] - v_amt; end if; if arr_net[i] > -v_eps then i := i + 1; end if; if arr_net[j] < v_eps then j := j - 1; end if; end loop; end if; drop table tmp_bals; return jsonb_build_object( 'total_cost', round(v_total,2), 'balances', v_balances, 'transfers', v_transfers ); end; $$;
